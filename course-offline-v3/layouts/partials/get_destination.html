{{/* Markdown link destinations for the offline v3 package. Course links (/courses/...) point at the live site: the package only contains this one course, and even its own pages are reached through page_url.html rather than from raw markdown paths. They get the v3 /courses/o/ prefix and are then forced fully qualified, because relativeURLs: true would otherwise rewrite a root-relative URL into a dead in-package path. Everything else internal gets an explicit /index.html on its path, ahead of any #fragment or ?query, so it resolves over file://. Links with a URL scheme (http, https, mailto, ftp, ...) pass through untouched. */}} {{ $destination := .Destination }} {{/* A URL scheme (http, https, mailto, ftp, ...) means the link is not a path into the package. */}} {{ $isInternalLink := not (findRE `^[A-Za-z][A-Za-z0-9+.-]*:` $destination) }} {{ $isExternalCourseLink := strings.HasPrefix $destination "/courses" }} {{ if $isInternalLink }} {{ if $isExternalCourseLink }} {{ $destination = partial "absolutize_course_url.html" (partial "canonical_course_url.html" $destination) }} {{ else }} {{ $destination = partial "explicit_index_url.html" $destination }} {{ end }} {{ end }} {{ return $destination }}
