{{/* Return the mirror drive home page, relative to each page. The mirror holds each course at its site_url_path (e.g. courses//), so its home page is that many levels above the package root. The URL stays root-relative so Hugo's relativeURLs post-processor prefixes each page's path to the package root; the "./" is needed because a bare "/../" is not rewritten correctly. A downloaded zip has no mirror home above it, so mirror_home_link.ts swaps in the live site there. */}} {{ $siteUrlPath := site.Data.course.site_url_path | default "" | strings.TrimPrefix "/" | strings.TrimSuffix "/" }} {{ $homeUrl := partial "site_root_url.html" "/" }} {{ if $siteUrlPath }} {{ $depth := len (split $siteUrlPath "/") }} {{ $homeUrl = printf "/./%sindex.html" (strings.Repeat $depth "../") }} {{ end }} {{ return $homeUrl }}
