{{/* Adds the v3 /courses/o/ prefix to an OCW course URL, preserving how the author scoped it: /courses//... -> /courses/o//... https://ocw.mit.edu/courses// -> https:///courses/o// A root-relative link stays root-relative, so it resolves against whichever host serves the build (production, RC, local, offline preview) and keeps the same-tab, same-origin behaviour it had under v2. Only an already fully qualified URL gets a host, and then it moves to COURSE_V3_CANONICAL_DOMAIN because the v2 host does not serve /courses/o/. Query strings and fragments are preserved. Anything that is not an OCW course path is returned untouched: other hosts, non-course paths such as /terms, the /courses/ listing itself, and URLs already under /courses/o/. COURSE_V3_CANONICAL_DOMAIN is only allowlisted in security.funcs.getenv for the ocw-course-v3 configs, which is why this lives in course-v3 rather than base-theme. */}} {{- $input := . -}} {{- $url := urls.Parse $input -}} {{- $result := $input -}} {{- $isOcwHost := or (eq $url.Host "") (in $url.Host "ocw.mit.edu") -}} {{- $slugAndRest := strings.TrimPrefix "/courses/" $url.Path -}} {{- $isCoursePath := and (strings.HasPrefix $url.Path "/courses/") (gt (len $slugAndRest) 0) (not (or (eq $url.Path "/courses/o") (strings.HasPrefix $url.Path "/courses/o/"))) -}} {{- if and $isOcwHost $isCoursePath -}} {{- $result = printf "/courses/o/%s" $slugAndRest -}} {{/* Authored links overwhelmingly omit Hugo's canonical trailing slash, which costs a redirect per link and leaves the same page reachable at two URLs (resource_link output is already slashed). Add it, unless the target looks like a file. */}} {{- if not (or (strings.HasSuffix $result "/") (findRE "\\.[a-zA-Z0-9]{2,5}$" $result)) -}} {{- $result = printf "%s/" $result -}} {{- end -}} {{- with $url.Host -}} {{/* Authored as a fully qualified URL, so keep it one — on the domain that actually serves /courses/o/. */}} {{- $result = printf "https://%s%s" (partial "course_v3_canonical_domain.html") $result -}} {{- end -}} {{- with $url.RawQuery -}} {{- $result = printf "%s?%s" $result . -}} {{- end -}} {{- with $url.Fragment -}} {{- $result = printf "%s#%s" $result . -}} {{- end -}} {{- end -}} {{- return $result -}}
