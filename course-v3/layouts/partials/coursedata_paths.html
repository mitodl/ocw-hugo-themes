{{/* Returns a copy of a page's or resource's params with the authored file paths mapped to where the v3 build serves them (see coursedata_path.html), for the data.json output. The v3 HTML makes the same correction when it renders these fields; without this, data.json would keep the v2 paths Studio wrote. Mapped fields: file video_files.video_thumbnail_file video_files.video_captions_file (legacy) video_files.video_transcript_file (legacy) video_files.video_captions_resources[].file video_files.video_transcript_resources[].file Every other key is returned unchanged. Hugo's merge lets the right-hand map win and merges nested maps key by key, but replaces slices, so the resource lists are rebuilt in full. */}} {{- $params := . -}} {{- $mapped := dict -}} {{- with $params.file -}} {{- $mapped = merge $mapped (dict "file" (partial "coursedata_path.html" .)) -}} {{- end -}} {{- with $params.video_files -}} {{- if reflect.IsMap . -}} {{- $videoFiles := . -}} {{- $mappedVideoFiles := dict -}} {{- range $key := slice "video_thumbnail_file" "video_captions_file" "video_transcript_file" -}} {{- with index $videoFiles $key -}} {{- $mappedVideoFiles = merge $mappedVideoFiles (dict $key (partial "coursedata_path.html" .)) -}} {{- end -}} {{- end -}} {{- range $key := slice "video_captions_resources" "video_transcript_resources" -}} {{- $resources := index $videoFiles $key -}} {{- if reflect.IsSlice $resources -}} {{- $mappedResources := slice -}} {{- range $resources -}} {{- $resource := . -}} {{- if reflect.IsMap $resource -}} {{- with $resource.file -}} {{- $resource = merge $resource (dict "file" (partial "coursedata_path.html" .)) -}} {{- end -}} {{- end -}} {{- $mappedResources = $mappedResources | append $resource -}} {{- end -}} {{- $mappedVideoFiles = merge $mappedVideoFiles (dict $key $mappedResources) -}} {{- end -}} {{- end -}} {{- with $mappedVideoFiles -}} {{- $mapped = merge $mapped (dict "video_files" .) -}} {{- end -}} {{- end -}} {{- end -}} {{- $result := $params -}} {{- with $mapped -}} {{- $result = merge $params . -}} {{- end -}} {{- return $result -}}
